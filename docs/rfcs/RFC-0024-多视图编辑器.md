# RFC-0024: 多视图编辑器（NL / SimpleTS / 表格）

> **状态**：DRAFT · **优先级**：P0 · **预计工作量**：5d · **阶段**：S6

---

## 1. 摘要

实现 orule-web 的多视图编辑器，支持 NL / SimpleTS / 表格 三视图切换（SimpleTS = 真值，其他视图派生）。

---

## 2. 动机

- 多视图编辑器是 orule-web 核心 UI（依据 `02-用例视图 §2.4.3`）
- SimpleTS = 唯一的语义中间态（依据 ADR-009），三视图同步围绕 SimpleTS 进行
- 依据 `06-运行视图 §6.1.2` 四视图编辑器的同步规则

---

### 3.2 编辑器状态

```typescript
// orule-web/src/components/RuleEditor/types.ts

export type EditorView = 'nl' | 'simplets' | 'table';

export interface EditorState {
  /** 当前活跃视图 */
  activeView: EditorView;
  
  /** NL 源码（用户输入） */
  nlSource: string;
  
  /** SimpleTS 源码（真值） */
  simpleTsSource: string;
  
  /** 表格结构 */
  tableData: TableRule | null;
  
  /** SimpleTS 是否只读 */
  isSimpleTsReadonly: boolean;
  
  /** 校验错误 */
  validationErrors: CompileError[];
  
  /** LLM 转换中 */
  isConverting: boolean;
  
  /** 是否已保存（脏标记） */
  isDirty: boolean;
}

export interface CompileError {
  line: number;
  column: number;
  message: string;
}
```

### 3.3 主容器（同步逻辑）

```tsx
// orule-web/src/components/RuleEditor/RuleEditor.tsx

export function RuleEditor({ ruleId }: { ruleId: string }) {
  const [state, setState] = useState<EditorState>({
    activeView: 'nl',
    nlSource: '',
    simpleTsSource: '',
    tableData: null,
    isSimpleTsReadonly: true,
    validationErrors: [],
    isConverting: false,
    isDirty: false,
  });

  // ========== 视图同步（核心逻辑）==========
  
  /**
   * NL → SimpleTS（LLM 转换）
   */
  const handleNlConvert = useCallback(async (nl: string) => {
    setState(s => ({ ...s, isConverting: true }));
    try {
      const result = await api.post('/api/v1/convert/nl-to-simplets', {
        ruleId,
        naturalLanguage: nl,
      });
      
      if (result.success) {
        setState(s => ({
          ...s,
          simpleTsSource: result.simpleTs,
          isDirty: true,
          isConverting: false,
          activeView: 'simplets',  // 自动切换到 SimpleTS
          validationErrors: [],
        }));
        // 同时更新表格视图
        await syncToTable(result.simpleTs);
      } else {
        // 失败时仍显示 partial 结果
        setState(s => ({
          ...s,
          simpleTsSource: result.simpleTs ?? s.simpleTsSource,
          isConverting: false,
          validationErrors: parseErrors(result.errorMessage),
        }));
        showErrorToast(result.errorMessage);
      }
    } catch (e) {
      setState(s => ({ ...s, isConverting: false }));
      showErrorToast('LLM 调用失败');
    }
  }, [ruleId]);

  /**
   * SimpleTS → NL（反向生成）
   */
  const handleSimpleTsToNl = useCallback(async (simplets: string) => {
    // 调用 LLM 将 SimpleTS 翻译回自然语言（可选功能）
    // MVP 暂不实现
  }, []);

  /**
   * SimpleTS → 表格
   */
  const handleSimpleTsToTable = useCallback(async (simplets: string) => {
    const table = await api.post('/api/v1/convert/simplets-to-table', { simpleTs: simplets });
    setState(s => ({ ...s, tableData: table }));
  }, []);

  /**
   * 表格 → SimpleTS
   */
  const handleTableToSimpleTs = useCallback(async (table: TableRule) => {
    const simpleTs = await api.post('/api/v1/convert/table-to-simplets', { table });
    setState(s => ({
      ...s,
      simpleTsSource: simpleTs,
      isDirty: true,
    }));
    await validateSimpleTs(simpleTs);
  }, []);

  /**
   * SimpleTS → TS-AST（只读，供调试）
   */
  const handleCompile = useCallback(async (simplets: string) => {
    const result = await api.post('/api/v1/rule-versions/' + currentVersionId + '/compile');
    if (result.success) {
      showSuccessToast('编译成功');
      await validateSimpleTs(simplets);
    } else {
      showErrorToast('编译失败: ' + result.errorMessage);
    }
  }, [currentVersionId]);

  /**
   * 保存（SimpleTS 为真值）
   */
  const handleSave = useCallback(async () => {
    await api.put('/api/v1/rule-versions/' + currentVersionId, {
      simpleTs: state.simpleTsSource,
    });
    setState(s => ({ ...s, isDirty: false }));
    showSuccessToast('保存成功');
  }, [currentVersionId, state.simpleTsSource]);

  return (
    <div className="rule-editor">
      <EditorToolbar
        activeView={state.activeView}
        isDirty={state.isDirty}
        isConverting={state.isConverting}
        hasErrors={state.validationErrors.length > 0}
        onTabChange={tab => setState(s => ({ ...s, activeView: tab }))}
        onConvert={handleNlConvert}
        onSave={handleSave}
        onCompile={handleCompile}
      />

      {/* 错误提示 */}
      {state.validationErrors.length > 0 && (
        <ValidationErrors errors={state.validationErrors} />
      )}

      {/* 视图切换 */}
      <Tabs activeKey={state.activeView} onChange={key => {
        if (key === 'table' && state.tableData === null) {
          handleSimpleTsToTable(state.simpleTsSource);
        }
        setState(s => ({ ...s, activeView: key as EditorView }));
      }}>
        <Tabs.Tab key="nl" tab="自然语言">
          <NlEditorView
            value={state.nlSource}
            onChange={nl => setState(s => ({ ...s, nlSource: nl }))}
            onConvert={handleNlConvert}
            isConverting={state.isConverting}
          />
        </Tabs.Tab>
        
        <Tabs.Tab key="simplets" tab="SimpleTS">
          <SimpleTsEditorView
            value={state.simpleTsSource}
            onChange={simplets => setState(s => ({ ...s, simpleTsSource: simplets, isDirty: true }))}
            readOnly={state.isSimpleTsReadonly}
            onValidate={validateSimpleTs}
          />
        </Tabs.Tab>
        
        <Tabs.Tab key="table" tab="表格">
          <TableEditorView
            data={state.tableData}
            onChange={handleTableToSimpleTs}
          />
        </Tabs.Tab>
        
        <Tabs.Tab key="ast" tab="TS-AST" disabled>
          <span style={{ color: '#999' }}>只读（调试用）</span>
        </Tabs.Tab>
        
        <Tabs.Tab key="groovy" tab="Groovy" disabled>
          <span style={{ color: '#999' }}>只读（编译产物）</span>
        </Tabs.Tab>
      </Tabs>

      {/* 测试用例面板（底部） */}
      <TestCasePanel ruleId={ruleId} />
    </div>
  );
}
```

### 3.4 工具栏

```tsx
// orule-web/src/components/RuleEditor/EditorToolbar.tsx

export function EditorToolbar({
  activeView, isDirty, isConverting, hasErrors,
  onTabChange, onConvert, onSave, onCompile
}: ToolbarProps) {
  return (
    <div className="editor-toolbar">
      {/* 保存按钮 */}
      <Button
        type="primary"
        disabled={!isDirty}
        onClick={onSave}
        icon={<SaveIcon />}
      >
        保存
      </Button>

      {/* 当前视图的专属按钮 */}
      {activeView === 'nl' && (
        <Button
          type="default"
          loading={isConverting}
          onClick={() => onConvert()}  // 从 context 获取 nl 值
          icon={<SparklesIcon />}
        >
          {isConverting ? 'LLM 转换中...' : 'LLM 转换 → SimpleTS'}
        </Button>
      )}

      {activeView === 'simplets' && (
        <Button
          type="default"
          disabled={isDirty}
          onClick={onCompile}
          icon={<PlayIcon />}
        >
          编译
        </Button>
      )}

      {/* 错误指示 */}
      {hasErrors && (
        <Tag color="error" icon={<WarningIcon />}>
          {activeView === 'simplets' ? '编译错误' : '校验失败'}
        </Tag>
      )}

      {/* 保存指示 */}
      {isDirty && <Tag color="warning">未保存</Tag>}
    </div>
  );
}
```
